CREATE TABLE "AMIUWebUtilities".trasporti_sap (
	id int4 GENERATED ALWAYS AS IDENTITY( INCREMENT BY 1 MINVALUE 1 MAXVALUE 2147483647 START 1 CACHE 1 NO CYCLE) NOT NULL,
	data_ins date NOT NULL,
	utente_ins varchar NOT NULL,
	num_trasp varchar NOT NULL,
	ticket varchar NULL,
	nota varchar NULL,
	CONSTRAINT pk_trasporti_sap PRIMARY KEY (id)
);
